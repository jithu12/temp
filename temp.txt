import uuid
from types import SimpleNamespace
from unittest.mock import Mock, patch, MagicMock

import pytest

from dataviz_api.apis.v1 import accounts_controller


# A fixed UUID we use as the "admin caller" in every test.
ALLOWED_ADMIN = uuid.UUID("00000000-0000-0000-0000-000000000001")
ALLOWED_SET = {ALLOWED_ADMIN}


def _make_account(status="ACTIVE", account_id=None):
    """
    Helper to build a SimpleNamespace that looks like an AccountDetails
    object. Status can be a string or an enum-like object with .value.
    """
    return SimpleNamespace(
        id=account_id or uuid.uuid4(),
        owner_account_id=uuid.uuid4(),
        status=SimpleNamespace(value=status),
        name="test-account",
    )


@pytest.mark.component
class TestV1Accounts:

    def setup_method(self):
        self.target_account_id = uuid.uuid4()
        self.caller_account_id = ALLOWED_ADMIN

    # ------------------------------------------------------------------
    # account_deactivate — happy path
    # ------------------------------------------------------------------
    @patch(
        "dataviz_api.apis.v1.accounts_controller.ALLOWED_ACCOUNT_IDS",
        ALLOWED_SET,
    )
    @patch("dataviz_api.apis.v1.accounts_controller.get_core")
    @patch("dataviz_api.apis.v1.accounts_controller.get_current_account_id")
    def test_account_deactivate(self, mock_caller, mock_get_core):
        mock_caller.return_value = self.caller_account_id

        mock_core = Mock()
        mock_get_core.return_value = mock_core

        # Pre-flight check calls get_by_owner_id — return ACTIVE account
        active_account = _make_account(status="ACTIVE")
        mock_core.accounts.get_by_owner_id.return_value = active_account

        # Core returns updated account after deactivation
        deactivated_account = _make_account(status="INACTIVE")
        mock_core.accounts.request_account_deactivation.return_value = (
            deactivated_account
        )

        response, status = accounts_controller.account_deactivate(
            target_account_id=str(self.target_account_id),
            account_id=str(self.caller_account_id),
        )

        assert status == 202
        assert response["status"] == "INACTIVE"
        mock_core.accounts.request_account_deactivation.assert_called_once_with(
            owner_account_id=self.target_account_id,
            account_id=self.caller_account_id,
        )

    # ------------------------------------------------------------------
    # account_deactivate — account already inactive → 409
    # ------------------------------------------------------------------
    @patch(
        "dataviz_api.apis.v1.accounts_controller.ALLOWED_ACCOUNT_IDS",
        ALLOWED_SET,
    )
    @patch("dataviz_api.apis.v1.accounts_controller.get_core")
    @patch("dataviz_api.apis.v1.accounts_controller.get_current_account_id")

        mock_caller.return_value = self.caller_account_id

        mock_core = Mock()
        mock_get_core.return_value = mock_core

    def test_account_status_not_found(self, mock_get_core):
        mock_core = Mock()
        mock_get_core.return_value = mock_core

        class AccountNotFoundException(Exception):
            pass

        mock_core.accounts.get_by_owner_id.side_effect = AccountNotFoundException(
            "no such account"
        )

        response, status = accounts_controller.account_status(
            target_account_id=str(self.target_account_id),
        )

        assert status == 404
        # New friendly message — check the code not the raw exception message
        assert response["code"] == "ACCOUNT_NOT_FOUND"
