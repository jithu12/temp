import uuid
from types import SimpleNamespace
from unittest.mock import Mock, patch

import pytest

from dataviz_api.apis.v1 import accounts_controller


# A fixed UUID we use as the "admin caller" in every test. We patch
# ALLOWED_ACCOUNT_IDS to contain this value so tests do not depend on
# the production hardcoded admin UUIDs.
ALLOWED_ADMIN = uuid.UUID("00000000-0000-0000-0000-000000000001")
ALLOWED_SET = {ALLOWED_ADMIN}


@pytest.mark.component
class TestV1Accounts:

    def setup_method(self):
        self.target_account_id = uuid.uuid4()
        self.caller_account_id = ALLOWED_ADMIN

    # ------------------------------------------------------------------
    # account_deactivate
    # ------------------------------------------------------------------
    @patch(
        "dataviz_api.apis.v1.accounts_controller.ALLOWED_ACCOUNT_IDS",
        ALLOWED_SET,
    )
    @patch("dataviz_api.apis.v1.accounts_controller.get_core")
    @patch("dataviz_api.apis.v1.accounts_controller.get_current_account_id")
    def test_account_deactivate(self, mock_caller, mock_get_core):
        mock_caller.return_value = self.caller_account_id

        mock_core = Mock()
        mock_get_core.return_value = mock_core

        returned_account = SimpleNamespace(
            id=self.target_account_id,
            owner_account_id=self.target_account_id,
            status="INACTIVE",
            name="test-account",
        )
        mock_core.accounts.request_account_deactivation.return_value = returned_account

        response, status = accounts_controller.account_deactivate(
            target_account_id=str(self.target_account_id),
        )

        assert status == 202
        assert response["id"] == str(self.target_account_id)
        assert response["status"] == "INACTIVE"
        mock_core.accounts.request_account_deactivation.assert_called_once_with(
            owner_account_id=self.target_account_id,
            account_id=self.caller_account_id,
        )

    # ------------------------------------------------------------------
    # account_deactivate -- when Core raises an exception, we surface 500
    # ------------------------------------------------------------------
    @patch(
        "dataviz_api.apis.v1.accounts_controller.ALLOWED_ACCOUNT_IDS",
        ALLOWED_SET,
    )
    @patch("dataviz_api.apis.v1.accounts_controller.get_core")
    @patch("dataviz_api.apis.v1.accounts_controller.get_current_account_id")
    def test_account_deactivate_with_exception(self, mock_caller, mock_get_core):
        mock_caller.return_value = self.caller_account_id

        mock_core = Mock()
        mock_get_core.return_value = mock_core
        mock_core.accounts.request_account_deactivation.side_effect = Exception("boom")

        response, status = accounts_controller.account_deactivate(
            target_account_id=str(self.target_account_id),
        )

        assert status == 500
        assert response["error"] == "boom"

    # ------------------------------------------------------------------
    # account_deactivate -- non-admin caller is rejected with 403
    # ------------------------------------------------------------------
    @patch(
        "dataviz_api.apis.v1.accounts_controller.ALLOWED_ACCOUNT_IDS",
        ALLOWED_SET,
    )
    @patch("dataviz_api.apis.v1.accounts_controller.get_core")
    @patch("dataviz_api.apis.v1.accounts_controller.get_current_account_id")
    def test_account_deactivate_non_admin_forbidden(
        self, mock_caller, mock_get_core
    ):
        # A caller UUID that is NOT in ALLOWED_ACCOUNT_IDS
        mock_caller.return_value = uuid.uuid4()

        mock_core = Mock()
        mock_get_core.return_value = mock_core

        # abort(403) raises an HTTPException; capture it
        from werkzeug.exceptions import Forbidden
        with pytest.raises(Forbidden):
            accounts_controller.account_deactivate(
                target_account_id=str(self.target_account_id),
            )

        # And Core must NOT have been called
        mock_core.accounts.request_account_deactivation.assert_not_called()

    # ------------------------------------------------------------------
    # account_status -- happy path (no admin gate)
    # ------------------------------------------------------------------
    @patch("dataviz_api.apis.v1.accounts_controller.get_core")
    def test_account_status(self, mock_get_core):
        mock_core = Mock()
        mock_get_core.return_value = mock_core

        mock_core.accounts.get_by_owner_id.return_value = SimpleNamespace(
            id=self.target_account_id,
            owner_account_id=self.target_account_id,
            status="ACTIVE",
            name="test-account",
        )

        response, status = accounts_controller.account_status(
            target_account_id=str(self.target_account_id),
        )

        assert status == 200
        assert response["id"] == str(self.target_account_id)
        assert response["status"] == "ACTIVE"
        assert response["name"] == "test-account"
        mock_core.accounts.get_by_owner_id.assert_called_once_with(
            str(self.target_account_id)
        )

    # ------------------------------------------------------------------
    # account_status -- when Status is an enum-like object with .value
    # ------------------------------------------------------------------
    @patch("dataviz_api.apis.v1.accounts_controller.get_core")
    def test_account_status_with_native_status(self, mock_get_core):
        mock_core = Mock()
        mock_get_core.return_value = mock_core

        # SimpleNamespace with a status that has .value (mimics an enum)
        enum_like_status = SimpleNamespace(value="SUSPENDED")
        mock_core.accounts.get_by_owner_id.return_value = SimpleNamespace(
            id=self.target_account_id,
            owner_account_id=self.target_account_id,
            status=enum_like_status,
            name="test-account",
        )

        response, status = accounts_controller.account_status(
            target_account_id=str(self.target_account_id),
        )

        assert status == 200
        assert response["status"] == "SUSPENDED"

    # ------------------------------------------------------------------
    # account_status -- account not found returns 404
    # ------------------------------------------------------------------
    @patch("dataviz_api.apis.v1.accounts_controller.get_core")
    def test_account_status_not_found(self, mock_get_core):
        mock_core = Mock()
        mock_get_core.return_value = mock_core

        # Build an exception class whose __name__ matches what
        # _handle_core_exception looks for.
        class AccountNotFoundException(Exception):
            pass

        mock_core.accounts.get_by_owner_id.side_effect = AccountNotFoundException(
            "no such account"
        )

        response, status = accounts_controller.account_status(
            target_account_id=str(self.target_account_id),
        )

        assert status == 404
        assert "no such account" in response["error"]

    # ------------------------------------------------------------------
    # account_status -- bad UUID in path param returns 400
    # ------------------------------------------------------------------
    @patch("dataviz_api.apis.v1.accounts_controller.get_core")
    def test_account_status_bad_uuid(self, mock_get_core):
        mock_core = Mock()
        mock_get_core.return_value = mock_core

        response, status = accounts_controller.account_status(
            target_account_id="not-a-uuid",
        )

        assert status == 400
        assert "Invalid UUID" in response["error"]
        mock_core.accounts.get_by_owner_id.assert_not_called()
